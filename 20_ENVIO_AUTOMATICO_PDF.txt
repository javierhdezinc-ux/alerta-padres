ENVÍO AUTOMÁTICO MENSUAL — sin trabajo extra del maestro
Enseñanza Digna · piloto

GitHub Pages NO envía correo. El robot es Google (una hoja + un script).
El maestro solo carga el aviso. El día 1 del mes Google manda:

  • Al padre: PDF / tabla de asistencia, conducta y felicitación de SU hijo
  • Al maestro: PDF / tabla de TODOS sus alumnos del mes
  • Copia a dirección (opcional)

function enviarResumenesMensuales() {
  var ss = SpreadsheetApp.getActive();
  var avisos = ss.getSheetByName("Hoja 1"); // o el nombre real de la hoja de avisos
  var datos = avisos.getDataRange().getValues();
  var ahora = new Date();
  var mes = ahora.getMonth();
  var year = ahora.getFullYear();
  // Si corres el día 1, el resumen es del MES ANTERIOR:
  var ref = new Date(year, mes, 0); // último día del mes pasado
  var m = ref.getMonth();
  var y = ref.getFullYear();

  var porPadre = {};
  var porMaestro = {};

  for (var i = 1; i < datos.length; i++) {
    var f = datos[i][0];
    if (!(f instanceof Date)) continue;
    if (f.getMonth() !== m || f.getFullYear() !== y) continue;
    var alumno = String(datos[i][1] || "").trim();
    var grado = datos[i][2];
    var materia = datos[i][3];
    var maestro = String(datos[i][4] || "").trim();
    var correoM = String(datos[i][5] || "").trim();
    var tutor = datos[i][6];
    var correoP = String(datos[i][7] || "").trim();
    var tipo = String(datos[i][8] || "");
    var hecho = datos[i][9];
    var linea = Utilities.formatDate(f, "America/Mexico_City", "dd/MM HH:mm")
      + " · " + tipo + " · " + materia + " · " + hecho;

    if (correoP) {
      if (!porPadre[correoP]) porPadre[correoP] = {tutor: tutor, alumno: alumno, lineas: []};
      porPadre[correoP].lineas.push(linea);
    }
    if (correoM) {
      if (!porMaestro[correoM]) porMaestro[correoM] = {maestro: maestro, lineas: []};
      porMaestro[correoM].lineas.push(alumno + " · " + linea);
    }
  }

  var mesTxt = Utilities.formatDate(ref, "America/Mexico_City", "MMMM yyyy");

  Object.keys(porPadre).forEach(function (mail) {
    var p = porPadre[mail];
    MailApp.sendEmail({
      to: mail,
      subject: "Enseñanza Digna · resumen " + mesTxt + " · " + p.alumno,
      body:
        "Tutor: " + p.tutor + "\nAlumno: " + p.alumno + "\nMes: " + mesTxt + "\n\n" +
        p.lineas.join("\n") +
        "\n\nEstos avisos no se borran. Usted marca Ya lo vi en la app.\n" +
        "Esto no es un chat."
    });
  });

  Object.keys(porMaestro).forEach(function (mail) {
    var t = porMaestro[mail];
    MailApp.sendEmail({
      to: mail,
      subject: "Enseñanza Digna · su relación " + mesTxt,
      body:
        "Maestro: " + t.maestro + "\nMes: " + mesTxt + "\n\n" +
        "Relación por alumno (asistencia, conducta, felicitación):\n\n" +
        t.lineas.join("\n") +
        "\n\nEsto llega solo. Usted no tiene que armar PDF."
    });
  });
}

function crearDisparadorDia1() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger("enviarResumenesMensuales")
    .timeBased()
    .onMonthDay(1)
    .atHour(7)
    .inTimezone("America/Mexico_City")
    .create();
}
